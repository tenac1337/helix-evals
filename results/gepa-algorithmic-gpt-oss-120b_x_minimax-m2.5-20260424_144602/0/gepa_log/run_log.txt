struct Polyomino {
    vector<pair<int, int>> cells; // cells in local coordinates
    int size;
    vector<vector<pair<int, int>>> all_transforms; // all 8 transformations (4 rotations * 2 reflections)

    Polyomino(const vector<pair<int, int>>& input_cells) : cells(input_cells), size(input_cells.size()) {
        generate_all_transforms();
    }

    void generate_all_transforms() {
        all_transforms.clear();
        vector<pair<int, int>> current = cells;

        // Generate 4 rotations
        for (int r = 0; r < 4; ++r) {
            vector<pair<int, int>> rotated;
            for (auto& p : current) {
                rotated.push_back({p.first, p.second});
            }
            all_transforms.push_back(rotated);
            // Rotate 90 degrees clockwise
            for (auto& p : current) {
                int x = p.first, y = p.second;
                p.first = y;
                p.second = -x;
            }
        }

        // Generate reflections of each rotation
        int sz = all_transforms.size();
        for (int i = 0; i < sz; ++i) {
            vector<pair<int, int>> reflected = all_transforms[i];
            for (auto& p : reflected) {
                p.first = -p.first;
            }
            all_transforms.push_back(reflected);
        }
    }

    // Get the bounding box of the polyomino in local coordinates
    pair<int, int> get_min_max() const {
        int min_x = INT_MAX, max_x = INT_MIN;
        int min_y = INT_MAX, max_y = INT_MIN;
        for (auto& p : cells) {

            max_x = max(max_x, p.first);

// Check if a polyomino can be placed at (x,y) with given rotation and reflection
bool can_place(const vector<pair<int, int>>& cells, int x, int y, int rot, int reflect, int W, int H) {
    vector<pair<int, int>> transformed;
    for (auto& p : cells) {
        int px = p.first, py = p.second;
        if (reflect) px = -px;
        // Apply rotation
        for (int i = 0; i < rot; ++i) {
            int nx = py, ny = -px;
            px = nx;
            py = ny;
        }
        transformed.push_back({px + x, py + y});
    }

    for (auto& p : transformed) {
        if (p.first < 0 || p.first >= W || p.second < 0 || p.second >= H) {
            return false;
        }
    }

    return true;
}

            }

}

    for (int i = 0; i < n; ++i) {

        bool placed = false;
        for (const auto& t : transforms) {
            int min_x = INT_MAX, max_x = INT_MIN, min_y = INT_MAX, max_y = INT_MIN;
            for (auto& p : t) {
                min_x = min(min_x, p.first);
                max_x = max(max_x, p.first);
                min_y = min(min_y, p.second);
                max_y = max(max_y, p.second);
            }

            int w = max_x - min_x + 1;
            int h = max_y - min_y + 1;

            // Try to place at (0,0) for now
            int x = 0, y = 0;
            bool valid = true;

            for (auto& p : t) {
                int px = p.first - min_x;
                int py = p.second - min_y;
                if (x + px >= W || y + py >= H || grid[x + px][y + py]) {
                    valid = false;
                    break;
                }
            }

            if (valid) {
                // Place the piece
                for (auto& p : t) {
                    int px = p.first - min_x;
                    int py = p.second - min_y;
                    grid[x + px][y + py] = true;
                }
                placements[i] = {x, y};
                placed = true;
                break;
            }
        }

        if (!placed) {
            // If not placed, increase size
            W = max(W, 1000);
            H = max(H, 1000);
            grid.assign(W, vector<bool>(H, false));
            i--; // Retry this piece
        }
    }

    // Output result
    cout << W << " " << H << "\n";
    for (int i = 0; i < n; ++i) {
        cout << placements[i].first << " " << placements[i].second << " 0 0\n";
    }
