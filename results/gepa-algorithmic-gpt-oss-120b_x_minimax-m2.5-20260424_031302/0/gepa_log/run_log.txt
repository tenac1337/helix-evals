// Function to generate all rotations and reflections of a polyomino
vector<vector<pair<int, int>>> generateTransformations(const vector<pair<int, int>>& cells) {
    vector<vector<pair<int, int>>> transformations;
    
    // 4 rotations + 4 reflections = 8 total transformations
    for (int rot = 0; rot < 4; ++rot) {
        vector<pair<int, int>> current = cells;
        // Rotate 90 degrees clockwise
        for (int i = 0; i < rot; ++i) {
            for (auto& p : current) {
                swap(p.first, p.second);
                p.second = -p.second;
            }
        }
        
        // Add original
        transformations.push_back(current);
        
        // Add reflection
        vector<pair<int, int>> reflected = current;
        for (auto& p : reflected) {
            p.first = -p.first;
        }
        transformations.push_back(reflected);
    }
    
    return transformations;
}

// Function to get bounding box of a polyomino
pair<pair<int, int>, pair<int, int>> getBoundingBox(const vector<pair<int, int>>& cells) {
    int minX = INT_MAX, maxX = INT_MIN;
    int minY = INT_MAX, maxY = INT_MIN;
    
    for (const auto& p : cells) {
        minX = min(minX, p.first);
        maxX = max(maxX, p.first);
        minY = min(minY, p.second);
        maxY = max(maxY, p.second);
    }
    
    return {{minX, minY}, {maxX, maxY}};
}

// Function to check if two polyominoes overlap
bool overlaps(const vector<pair<int, int>>& cells1, const vector<pair<int, int>>& cells2,
              int x1, int y1, int x2, int y2) {
    set<pair<int, int>> s1, s2;
    
    for (const auto& p : cells1) {
        s1.insert({p.first + x1, p.second + y1});
    }
    
    for (const auto& p : cells2) {
        s2.insert({p.first + x2, p.second + y2});
    }
    
    for (const auto& p : s1) {
        if (s2.count(p)) {
            return true;
        }
    }
    
    return false;
}

// Function to fit all polyominoes into a rectangle
bool fit(const vector<Polyomino>& polyominoes, int W, int H, vector<tuple<int, int, int, int>>& placements) {
    int n = polyominoes.size();
    vector<vector<vector<pair<int, int>>>> allTransformations(n);

                    if (backtrack(index + 1, currentPlacements)) {

// Function to calculate total area
long long totalArea(const vector<Polyomino>& polyominoes) {
    long long total = 0;
    for (const auto& p : polyominoes) {
        total += p.cells.size();
    }
    return total;
}
