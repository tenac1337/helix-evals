// Generate all possible rotations and reflections of a polyomino
vector<vector<pair<int, int>>> generateTransforms(const vector<pair<int, int>>& cells) {
    vector<vector<pair<int, int>>> transforms;
    
    // Original
    transforms.push_back(cells);
    
    // 90° clockwise
    vector<pair<int, int>> rot90(cells.size());
    for (int i = 0; i < cells.size(); ++i) {
        rot90[i] = {cells[i].second, -cells[i].first};
    }
    transforms.push_back(rot90);
    
    // 180°
    vector<pair<int, int>> rot180(cells.size());
    for (int i = 0; i < cells.size(); ++i) {
        rot180[i] = {-cells[i].first, -cells[i].second};
    }
    transforms.push_back(rot180);
    
    // 270° clockwise
    vector<pair<int, int>> rot270(cells.size());
    for (int i = 0; i < cells.size(); ++i) {
        rot270[i] = {-cells[i].second, cells[i].first};
    }
    transforms.push_back(rot270);
    
    // Reflect then rotate
    vector<pair<int, int>> reflect(cells.size());
    for (int i = 0; i < cells.size(); ++i) {
        reflect[i] = {-cells[i].first, cells[i].second};
    }
    transforms.push_back(reflect);
    
    // Reflect + 90°
    vector<pair<int, int>> r90(cells.size());
    for (int i = 0; i < cells.size(); ++i) {
        r90[i] = {cells[i].second, cells[i].first};
    }
    transforms.push_back(r90);
    
    // Reflect + 180°
    vector<pair<int, int>> r180(cells.size());
    for (int i = 0; i < cells.size(); ++i) {
        r180[i] = {cells[i].first, -cells[i].second};
    }
    transforms.push_back(r180);
    
    // Reflect + 270°
    vector<pair<int, int>> r270(cells.size());
    for (int i = 0; i < cells.size(); ++i) {
        r270[i] = {-cells[i].second, -cells[i].first};
    }
    transforms.push_back(r270);
    
    return transforms;
}

// Get bounding box of a set of cells
pair<int, int> getBounds(const vector<pair<int, int>>& cells) {
    int minX = INT_MAX, minY = INT_MAX, maxX = INT_MIN, maxY = INT_MIN;
    for (auto& p : cells) {
        minX = min(minX, p.first);
        minY = min(minY, p.second);
        maxX = max(maxX, p.first);
        maxY = max(maxY, p.second);
    }
    return {maxX - minX + 1, maxY - minY + 1};
}

// Try to place a polyomino at given position with given transform
bool canPlace(const vector<pair<int, int>>& cells, int x, int y, 
              const vector<pair<int, int>>& transformedCells, 
              vector<vector<bool>>& grid) {
    for (auto& p : transformedCells) {
        int nx = x + p.first;
        int ny = y + p.second;
        if (nx < 0 || ny < 0 || grid[nx][ny]) return false;
    }
    return true;
}

// Place polyomino
void place(const vector<pair<int, int>>& cells, int x, int y, 
           const vector<pair<int, int>>& transformedCells, 
           vector<vector<bool>>& grid) {
    for (auto& p : transformedCells) {
        int nx = x + p.first;
        int ny = y + p.second;
        grid[nx][ny] = true;
    }
}

// Remove polyomino
void remove(const vector<pair<int, int>>& cells, int x, int y, 
            const vector<pair<int, int>>& transformedCells, 
            vector<vector<bool>>& grid) {
    for (auto& p : transformedCells) {
        int nx = x + p.first;
        int ny = y + p.second;
        grid[nx][ny] = false;
    }
}
