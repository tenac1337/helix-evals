        for (auto& p : cells) {
            min_x = min(min_x, p.first);
            min_y = min(min_y, p.second);
        }
        for (auto& p : cells) {
            p.first -= min_x;
            p.second -= min_y;
        }
        width = 0, height = 0;
        for (auto& p : cells) {
            width = max(width, p.first);
            height = max(height, p.second);
        }
        width++;
        height++;
    }
    
    vector<vector<pair<int, int>>> get_transforms() {
        vector<vector<pair<int, int>>> transforms;
        vector<pair<int, int>> current = cells;
        
        // 4 rotations + 1 reflection + 4 rotations
        for (int i = 0; i < 2; i++) {
            for (int j = 0; j < 4; j++) {
                transforms.push_back(current);
                // Rotate 90 degrees clockwise
                for (auto& p : current) {
                    int x = p.first, y = p.second;
                    p.first = y;
                    p.second = -x;
                }
            }
            // Reflect across y-axis
            for (auto& p : current) {
                p.first = -p.first;
            }
        }
        return transforms;
    }
};
