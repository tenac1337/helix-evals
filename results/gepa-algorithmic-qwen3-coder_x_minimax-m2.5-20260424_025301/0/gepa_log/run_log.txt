using namespace std;

// Structure to represent a polyomino
struct Polyomino {
    vector<pair<int, int>> cells;
    int id;
    
    // Generate all possible rotations and reflections
    vector<vector<pair<int, int>>> getTransformations() {
        vector<vector<pair<int, int>>> transformations;
        set<vector<pair<int, int>>> uniqueTransforms;
        
        // For each of the 4 rotations and 2 reflections (8 total)
        for (int r = 0; r < 4; ++r) {
            vector<pair<int, int>> rotated = cells;
            for (int i = 0; i < r; ++i) {
                for (auto& p : rotated) {
                    swap(p.first, p.second);
                    p.second = -p.second;
                }
            }
            
            // Original
            uniqueTransforms.insert(rotated);
            // Reflected
            vector<pair<int, int>> reflected(rotated);
            for (auto& p : reflected) {
                p.first = -p.first;
            }
            uniqueTransforms.insert(reflected);
        }
        
        transformations.assign(uniqueTransforms.begin(), uniqueTransforms.end());
        return transformations;
    }
};

// Helper to find bounding box of a polyomino
pair<int, int> getBbox(const vector<pair<int, int>>& cells) {
    int minX = INT_MAX, minY = INT_MAX;
    int maxX = INT_MIN, maxY = INT_MIN;
    
    for (const auto& p : cells) {
        minX = min(minX, p.first);
        minY = min(minY, p.second);
        maxX = max(maxX, p.first);
        maxY = max(maxY, p.second);
    }
    
    return {maxX - minX + 1, maxY - minY + 1};
}

// Check if a placement is valid
bool isValidPlacement(const vector<pair<int, int>>& cells, int x, int y, 
                      int W, int H, vector<vector<bool>>& grid) {
    for (const auto& p : cells) {
        int nx = x + p.first;
        int ny = y + p.second;
        if (nx < 0 || nx >= W || ny < 0 || ny >= H) return false;
        if (grid[ny][nx]) return false;
    }
    return true;
}

// Place a polyomino on the grid
void placePolyomino(const vector<pair<int, int>>& cells, int x, int y, 
                    vector<vector<bool>>& grid) {
    for (const auto& p : cells) {
        int nx = x + p.first;
        int ny = y + p.second;
        grid[ny][nx] = true;
    }
}

// Remove a polyomino from the grid
void removePolyomino(const vector<pair<int, int>>& cells, int x, int y, 
                     vector<vector<bool>>& grid) {
    for (const auto& p : cells) {
        int nx = x + p.first;
        int ny = y + p.second;
        grid[ny][nx] = false;
    }
}

using namespace std;

struct Polyomino {
    vector<pair<int, int>> cells;
    int width, height;
    
    Polyomino() : width(0), height(0) {}
    
    void normalize() {
        if (cells.empty()) return;
        
        int min_x = INT_MAX, min_y = INT_MAX;
        for (auto& p : cells) {
            min_x = min(min_x, p.first);
            min_y = min(min_y, p.second);
        }
        
        for (auto& p : cells) {
            p.first -= min_x;
            p.second -= min_y;
        }
        
        width = 0, height = 0;
        for (auto& p : cells) {
            width = max(width, p.first);
            height = max(height, p.second);
        }
        width++;
        height++;
    }
    
    vector<vector<pair<int, int>>> get_transforms() {
        vector<vector<pair<int, int>>> transforms;
        
        // 4 rotations + 4 reflections (8 total)
        vector<vector<pair<int, int>>> shapes(8, cells);
        
        // rotations
        for (int i = 0; i < 4; i++) {
            for (auto& p : shapes[i]) {
                swap(p.first, p.second);
                p.second = -p.second;
            }
        }
        
        // reflections
        for (int i = 0; i < 4; i++) {
            shapes[i+4] = shapes[i];
            for (auto& p : shapes[i+4]) {
                p.first = -p.first;
            }
        }
        
        for (int i = 0; i < 8; i++) {
            transforms.push_back(shapes[i]);
        }
        
        return transforms;
    }
};
