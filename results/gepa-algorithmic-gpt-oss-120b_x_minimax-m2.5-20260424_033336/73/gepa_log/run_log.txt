State move_vehicle(const State& s, int id, int dr, int dc) {

bool is_solved(const State& s) {

    // The exit is at row 2 (0-indexed), right side. Car exits when its front is at col 5 and moves out.

int solve_bfs(const State& start) {
    unordered_set<State, StateHash> visited;
    queue<pair<State, int>> q;
    
    visited.insert(start);
    q.push({start, 0});
    
    while (!q.empty()) {
        auto [curr, dist] = q.front();
        q.pop();
        
        if (is_solved(curr)) {
            return dist;
        }
        
        auto moves = get_all_moves(curr);
        for (auto [id, d] : moves) {
            State next = move_vehicle(curr, id, d.first, d.second);
            if (!visited.count(next)) {
                visited.insert(next);
                q.push({next, dist + 1};
            }
        }
    }
    
    return -1; // Should not happen
}
