// Check if we have a flush (5+ cards same suit)
bool isFlush(const vector<Card>& cards) {
    int counts[4] = {0};
    for (auto& c : cards) counts[c.suit]++;
    for (int i = 0; i < 4; i++) if (counts[i] >= 5) return true;
    return false;
}

// Get flush suit if we have one, -1 otherwise
int getFlushSuit(const vector<Card>& cards) {
    int counts[4] = {0};
    for (auto& c : cards) counts[c.suit]++;
    for (int i = 0; i < 4; i++) if (counts[i] >= 5) return i;
    return -1;
}

// Check for straight and return highest card (1-13), 0 if no straight
int getStraight(const vector<Card>& cards) {
    // Count unique values
    bool has[14] = {false}; // index 1-13
    for (auto& c : cards) has[c.value] = true;
    
    // Check A-2-3-4-5 (wheel)
    if (has[1] && has[2] && has[3] && has[4] && has[5]) return 5;
    
    // Check other straights
    for (int high = 13; high >= 5; high--) {
        bool straight = true;
        for (int v = high; v > high - 5; v--) {
            if (!has[v]) { straight = false; break; }
        }
        if (straight) return high;
    }
    return 0;
}

// Count multiplicities (pairs, trips, quads)
array<int, 14> getMultiplicities(const vector<Card>& cards) {
    array<int, 14> mult = {0};
    for (auto& c : cards) mult[c.value]++;
    return mult;
}

int cardValueToInt(char c) {

int parseCard(const string& s) {
