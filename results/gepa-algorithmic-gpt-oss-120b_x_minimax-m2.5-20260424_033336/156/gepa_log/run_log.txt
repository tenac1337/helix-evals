void apply_move(vector<vector<int>>& grid, int& er, int& ec, char move) {
    if (move == 'U') {
        swap(grid[er][ec], grid[er-1][ec]);
        er--;
    } else if (move == 'D') {
        swap(grid[er][ec], grid[er+1][ec]);
        er++;
    } else if (move == 'L') {
        swap(grid[er][ec], grid[er][ec-1]);
        ec--;
    } else if (move == 'R') {
        swap(grid[er][ec], grid[er][ec+1]);
        ec++;
    }
}

bool is_valid_move(char move, int er, int ec, int N) {
    if (move == 'U') return er > 0;
    if (move == 'D') return er < N-1;
    if (move == 'L') return ec > 0;
    if (move == 'R') return ec < N-1;
    return false;
}

size_t hash_grid(const vector<vector<int>>& grid) {
    size_t seed = 0;
    for (const auto& row : grid) {
        for (int val : row) {
            seed ^= hash<int>()(val) + 0x9e3779b9 + (seed << 6) + (seed >> 2);
        }
    }
    return seed;
}
