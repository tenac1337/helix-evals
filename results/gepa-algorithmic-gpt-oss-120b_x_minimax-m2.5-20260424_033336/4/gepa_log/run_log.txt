Piece transform_piece(const Piece& p, int rot, bool reflect) {
    Piece res;
    res.cells.reserve(p.cells.size());
    for (auto [x, y] : p.cells) {
        int nx = x, ny = y;
        if (reflect) nx = -nx;
        // rotation: 0=0°, 1=90°, 2=180°, 3=270°
        if (rot == 1) { int t = nx; nx = -ny; ny = t; }
        else if (rot == 2) { nx = -nx; ny = -ny; }
        else if (rot == 3) { int t = nx; nx = ny; ny = -t; }
        res.cells.push_back({nx, ny});
    }
    // normalize to non-negative coordinates
    int minx = INT_MAX, miny = INT_MAX;
    for (auto& c : res.cells) {
        minx = min(minx, c.first);
        miny = min(miny, c.second);
    }
    for (auto& c : res.cells) {
        c.first -= minx;
        c.second -= miny;
    }
    // compute bounding box
    res.minx = 0; res.miny = 0;
    res.w = 0; res.h = 0;
    for (auto [x, y] : res.cells) {
        res.w = max(res.w, x + 1);
        res.h = max(res.h, y + 1);
    }
    return res;
}
