// Generate all possible rotations and reflections of a polyomino
vector<vector<pair<int, int>>> generateTransforms(const vector<pair<int, int>>& cells) {
    vector<vector<pair<int, int>>> transforms;
    
    // 4 rotations + 4 reflections = 8 total
    for (int rot = 0; rot < 4; ++rot) {
        vector<pair<int, int>> current = cells;
        // Rotate 90 degrees clockwise
        for (int i = 0; i < rot; ++i) {
            for (auto& p : current) {
                int x = p.first, y = p.second;
                p.first = y;
                p.second = -x;
            }
        }
        
        // Add the original
        transforms.push_back(current);
        
        // Add reflection
        vector<pair<int, int>> reflected = current;
        for (auto& p : reflected) {
            p.first = -p.first;
        }
        transforms.push_back(reflected);
    }
    
    return transforms;
}

}
