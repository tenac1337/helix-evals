struct Polyomino {
    vector<pair<int, int>> cells;
    int minx, miny, maxx, maxy;
    int width, height;
    
    Polyomino() : minx(0), miny(0), maxx(-1), maxy(-1), width(0), height(0) {}
    
    void init(const vector<pair<int, int>>& input_cells) {
        cells = input_cells;
        minx = miny = INT_MAX;
        maxx = maxy = INT_MIN;
        for (auto& p : cells) {
            minx = min(minx, p.first);
            miny = min(miny, p.second);
            maxx = max(maxx, p.first);
            maxy = max(maxy, p.second);
        }
        width = maxx - minx + 1;
        height = maxy - miny + 1;
    }
    
    vector<pair<int, int>> get_transformed_cells(int tx, int ty, int rot, bool flip) const {
        vector<pair<int, int>> result;
        for (auto& p : cells) {
            int x = p.first, y = p.second;
            if (flip) x = -x;
            for (int i = 0; i < rot; ++i) {
                int nx = y, ny = -x;
                x = nx; y = ny;
            }
            result.push_back({x + tx, y + ty});
        }
        return result;
    }
};

// Function to generate all possible rotations and reflections of a polyomino
vector<vector<pair<int,int>>> get_transforms(const vector<pair<int,int>>& cells) {
    vector<vector<pair<int,int>>> transforms;
    
    // 4 rotations + 4 reflections (8 total)
    vector<vector<pair<int,int>>> rotations;
    for (int i = 0; i < 4; ++i) {
        vector<pair<int,int>> rotated;
        for (auto& p : cells) {
            int x = p.first, y = p.second;
            if (i == 0) rotated.push_back({x, y});
            else if (i == 1) rotated.push_back({y, -x});
            else if (i == 2) rotated.push_back({-x, -y});
            else if (i == 3) rotated.push_back({-y, x});
        }
        rotations.push_back(rotated);
    }
    
    // Generate reflections
    vector<vector<pair<int,int>>> reflected;
    for (auto& r : rotations) {
        vector<pair<int,int>> refl;
        for (auto& p : r) {
            refl.push_back({-p.first, p.second});
        }
        reflected.push_back(refl);
    }
    
    // Combine rotations and reflections
    for (auto& r : rotations) transforms.push_back(r);
    for (auto& r : reflected) transforms.push_back(r);
    
    return transforms;
}

// Function to compute bounding box of a transformed polyomino
pair<pair<int,int>, pair<int,int>> get_bounding_box(const vector<pair<int,int>>& cells) {
    int min_x = INT_MAX, max_x = INT_MIN;
    int min_y = INT_MAX, max_y = INT_MIN;
    
    for (auto& p : cells) {
        min_x = min(min_x, p.first);
        max_x = max(max_x, p.first);
        min_y = min(min_y, p.second);
        max_y = max(max_y, p.second);
    }
    
    return {{min_x, max_x}, {min_y, max_y}};
}
