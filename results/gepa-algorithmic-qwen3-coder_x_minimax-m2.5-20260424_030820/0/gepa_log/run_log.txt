// Structure to represent a polyomino
struct Polyomino {
    vector<pair<int, int>> cells;
    int id;
    
    Polyomino(int _id) : id(_id) {}
    
    // Generate all possible rotations and reflections
    vector<vector<pair<int, int>>> getTransformations() {
        vector<vector<pair<int, int>>> transformations;
        
        // Original shape
        transformations.push_back(cells);
        
        // 90° clockwise rotation
        vector<pair<int, int>> rot90(cells.size());
        for (int i = 0; i < cells.size(); i++) {
            rot90[i] = {cells[i].second, -cells[i].first};
        }
        transformations.push_back(rot90);
        
        // 180° rotation
        vector<pair<int, int>> rot180(cells.size());
        for (int i = 0; i < cells.size(); i++) {
            rot180[i] = {-cells[i].first, -cells[i].second};
        }
        transformations.push_back(rot180);
        
        // 270° clockwise rotation
        vector<pair<int, int>> rot270(cells.size());
        for (int i = 0; i < cells.size(); i++) {
            rot270[i] = {-cells[i].second, cells[i].first};
        }
        transformations.push_back(rot270);
        
        // Reflection across y-axis
        vector<pair<int, int>> reflect(cells.size());
        for (int i = 0; i < cells.size(); i++) {
            reflect[i] = {-cells[i].first, cells[i].second};
        }
        transformations.push_back(reflect);
        
        // Reflection + 90°
        vector<pair<int, int>> reflect90(cells.size());
        for (int i = 0; i < cells.size(); i++) {
            reflect90[i] = {cells[i].second, cells[i].first};
        }
        transformations.push_back(reflect90);
        
        // Reflection + 180°

        }
        transformations.push_back(reflect270);
        
        return transformations;
    }
};

// Function to find bounding box of a set of points
pair<pair<int, int>, pair<int, int>> getBoundingBox(const vector<pair<int, int>>& points) {
    int minX = INT_MAX, minY = INT_MAX;
    int maxX = INT_MIN, maxY = INT_MIN;
    
    for (auto p : points) {
        minX = min(minX, p.first);
        minY = min(minY, p.second);
        maxX = max(maxX, p.first);
        maxY = max(maxY, p.second);
    }
    
    return {{minX, minY}, {maxX, maxY}};
}

// Function to check if two polyominoes overlap
bool overlaps(const vector<pair<int, int>>& p1, const vector<pair<int, int>>& p2) {
    set<pair<int, int>> s1(p1.begin(), p1.end());
    set<pair<int, int>> s2(p2.begin(), p2.end());
    
    for (auto& point : s1) {
        if (s2.count(point)) return true;
    }
    return false;
}

// Function to place a polyomino at a given position with rotation and reflection
vector<pair<int, int>> place(const vector<pair<int, int>>& cells, int x, int y, int rot, int reflect) {
    vector<pair<int, int>> result;
    vector<pair<int, int>> transformed = cells;
    
    // Apply reflection if needed
    if (reflect) {
        for (auto& p : transformed) {
            p.first = -p.first;
        }
    }
    
    // Apply rotation
    for (int i = 0; i < rot; i++) {
        for (auto& p : transformed) {
            int temp = p.first;
            p.first = p.second;
            p.second = -temp;
        }
    }
    
    // Apply translation
    for (auto& p : transformed) {
        p.first += x;
        p.second += y;
    }
    
    return transformed;
}

// Check if placing piece at (x, y) with given transform overlaps with existing pieces
bool can_place(const vector<pair<int, int>>& shape, int x, int y, 
               const vector<set<pair<int, int>>>& placed_cells) {
    for (const auto& p : shape) {
        int px = p.first + x;
        int py = p.second + y;
        for (const auto& placed : placed_cells) {
            if (placed.count({px, py})) return false;
        }
    }
    return true;
}

            for (const auto& p : shape) {
                min_x = min(min_x, p.first);

                max_y = max(max_y, p.second);

            for (int x = 0; x <= W - (max_x - min_x); ++x) {

                        for (const auto& p : shape) {

                        rotations[idx] = t / 4;

                        positions[idx] = {x, y};

                        placed_cells[idx].clear();
                    }
                }
            }
        }
        return false;
    };

}
