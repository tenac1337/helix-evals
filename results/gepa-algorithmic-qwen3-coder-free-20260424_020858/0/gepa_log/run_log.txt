            addFrontier(cx, cy - 1);

        curMinX = min(curMinX, bestX);
        curMaxX = max(curMaxX, bestX + (maxx - minx));
        curMinY = min(curMinY, bestY);
        curMaxY = max(curMaxY, bestY + (maxy - miny));
    }
    
    // Final bounding box
    long long finalW = curMaxX - curMinX + 1;
    long long finalH = curMaxY - curMinY + 1;

    }
