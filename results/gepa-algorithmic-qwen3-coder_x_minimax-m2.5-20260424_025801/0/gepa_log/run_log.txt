        for (auto& p : cells) {
            min_x = min(min_x, p.first);
            min_y = min(min_y, p.second);
        }
        for (auto& p : cells) {
            p.first -= min_x;
            p.second -= min_y;
        }
        width = 0, height = 0;
        for (auto& p : cells) {
            width = max(width, p.first);
            height = max(height, p.second);
        }
        width++;
        height++;
    }

            }
        }
        return transforms;
    }
};

// Function to generate all rotations and reflections of a polyomino
vector<vector<pair<int, int>>> generate_transforms(const vector<pair<int, int>>& cells) {
    vector<vector<pair<int, int>>> transforms;
    
    // 4 rotations + 4 reflections
    vector<vector<pair<int, int>>> base_transforms;
    
    // Original
    base_transforms.push_back(cells);
    
    // 90° clockwise
    vector<pair<int, int>> rot90;
    for (auto& p : cells) {
        rot90.push_back({p.second, -p.first});
    }
    base_transforms.push_back(rot90);
    
    // 180°
    vector<pair<int, int>> rot180;
    for (auto& p : cells) {
        rot180.push_back({-p.first, -p.second});
    }
    base_transforms.push_back(rot180);
    
    // 270° clockwise
    vector<pair<int, int>> rot270;
    for (auto& p : cells) {
        rot270.push_back({-p.second, p.first});
    }
    base_transforms.push_back(rot270);
    
    // Reflections
    vector<pair<int, int>> reflect;
    for (auto& p : cells) {
        reflect.push_back({-p.first, p.second});
    }
    base_transforms.push_back(reflect);
    
    // 90° + reflection
    vector<pair<int, int>> r90;
    for (auto& p : cells) {
        r90.push_back({p.second, p.first});
    }
    base_transforms.push_back(r90);
    
    // 180° + reflection
    vector<pair<int, int>> r180;
    for (auto& p : cells) {
        r180.push_back({p.first, -p.second});
    }
    base_transforms.push_back(r180);
    
    // 270° + reflection
    vector<pair<int, int>> r270;
    for (auto& p : cells) {
        r270.push_back({-p.second, -p.first});
    }
    base_transforms.push_back(r270);
    
    transforms = base_transforms;
    
    return transforms;
}

// Function to get bounding box of a transformed polyomino
pair<int, int> get_bounding_box(const vector<pair<int, int>>& cells) {
    int min_x = INT_MAX, max_x = INT_MIN;
    int min_y = INT_MAX, max_y = INT_MIN;
    
    for (auto& p : cells) {
        min_x = min(min_x, p.first);
        max_x = max(max_x, p.first);
        min_y = min(min_y, p.second);
        max_y = max(max_y, p.second);
    }
    
    return {max_x - min_x + 1, max_y - min_y + 1};
}

// Function to check if two polyominoes overlap
bool overlaps(const vector<pair<int, int>>& cells1, const vector<pair<int, int>>& cells2,
              int x1, int y1, int x2, int y2) {
    set<pair<int, int>> s1, s2;
    for (auto& p : cells1) {
        s1.insert({p.first + x1, p.second + y1});
    }
    for (auto& p : cells2) {
        s2.insert({p.first + x2, p.second + y2});
    }
    
    for (auto& p : s1) {
        if (s2.count(p)) return true;
    }
    return false;
}
