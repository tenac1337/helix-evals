// Generate all possible rotations and reflections of a polyomino
vector<vector<pair<int, int>>> generate_transforms(const vector<pair<int, int>>& cells) {
    vector<vector<pair<int, int>>> transforms;
    
    // 4 rotations + 4 reflections = 8 total
    vector<vector<pair<int, int>>> rotations;
    vector<pair<int, int>> temp = cells;
    
    // Generate 4 rotations
    for (int i = 0; i < 4; ++i) {
        rotations.push_back(temp);
        // Rotate 90 degrees clockwise
        for (auto& p : temp) {
            int x = p.first, y = p.second;
            p.first = y;
            p.second = -x;
        }
    }
    
    // Add reflections
    for (const auto& rot : rotations) {
        transforms.push_back(rot);
        vector<pair<int, int>> reflected = rot;
        for (auto& p : reflected) {
            p.first = -p.first;
        }
        transforms.push_back(reflected);
    }
    
    return transforms;
}

// Check if a polyomino can be placed at (x, y) with given rotation and reflection
bool can_place(const vector<pair<int, int>>& cells, int x, int y, const vector<pair<int, int>>& shape) {
    // Translate shape to position (x, y)
    for (const auto& p : shape) {
        int px = p.first + x;
        int py = p.second + y;
        if (find(cells.begin(), cells.end(), make_pair(px, py)) != cells.end()) {
            return false;
        }
    }
    return true;
}

// Get bounding box of a polyomino
pair<int, int> get_bounding_box(const vector<pair<int, int>>& cells) {
    int min_x = INT_MAX, max_x = INT_MIN;
    int min_y = INT_MAX, max_y = INT_MIN;
    
    for (const auto& p : cells) {
        min_x = min(min_x, p.first);
        max_x = max(max_x, p.first);
        min_y = min(min_y, p.second);
        max_y = max(max_y, p.second);
    }
    
    return {max_x - min_x + 1, max_y - min_y + 1};
}
