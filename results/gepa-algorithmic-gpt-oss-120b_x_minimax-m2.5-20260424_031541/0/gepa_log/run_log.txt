vector<pair<int,int>> rotate_cw(const vector<pair<int,int>>& cells) {
    vector<pair<int,int>> res;
    res.reserve(cells.size());
    for (auto &c : cells) {
        res.emplace_back(-c.second, c.first);
    }
    return res;
}

vector<pair<int,int>> reflect_y(const vector<pair<int,int>>& cells) {
    vector<pair<int,int>> res;
    res.reserve(cells.size());
    for (auto &c : cells) {
        res.emplace_back(-c.first, c.second);
    }
    return res;
}

vector<pair<int,int>> normalize(const vector<pair<int,int>>& cells, int &w, int &h) {
    int minx = cells[0].first, maxx = cells[0].first;
    int miny = cells[0].second, maxy = cells[0].second;
    for (auto &c : cells) {
        minx = min(minx, c.first);
        maxx = max(maxx, c.first);
        miny = min(miny, c.second);
        maxy = max(maxy, c.second);
    }
    w = maxx - minx + 1;
    h = maxy - miny + 1;
    vector<pair<int,int>> res;
    res.reserve(cells.size());
    for (auto &c : cells) {
        res.emplace_back(c.first - minx, c.second - miny);
    }
    return res;
}
