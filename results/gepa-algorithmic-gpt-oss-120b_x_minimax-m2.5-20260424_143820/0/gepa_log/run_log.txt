// Function to find bounding box of a set of points
pair<int, int> getMinBB(const vector<pair<int, int>>& points) {
    int minX = INT_MAX, minY = INT_MAX;
    for (const auto& p : points) {
        minX = min(minX, p.first);
        minY = min(minY, p.second);
    }
    return {minX, minY};
}

pair<int, int> getMaxBB(const vector<pair<int, int>>& points) {
    int maxX = INT_MIN, maxY = INT_MIN;
    for (const auto& p : points) {
        maxX = max(maxX, p.first);
        maxY = max(maxY, p.second);
    }
    return {maxX, maxY};
}

// Function to translate a shape to origin
vector<pair<int, int>> translateToOrigin(const vector<pair<int, int>>& shape) {
    auto minBB = getMinBB(shape);
    vector<pair<int, int>> translated(shape.size());
    for (int i = 0; i < shape.size(); i++) {
        translated[i] = {shape[i].first - minBB.first, shape[i].second - minBB.second};
    }
    return translated;
}

// Function to compute area of bounding box
int computeArea(const vector<pair<int, int>>& shape) {
    auto minBB = getMinBB(shape);
    auto maxBB = getMaxBB(shape);
    return (maxBB.first - minBB.first + 1) * (maxBB.second - minBB.second + 1);
}

// Function to check if two shapes overlap
bool checkOverlap(const vector<pair<int, int>>& shape1, const vector<pair<int, int>>& shape2,
                  int x1, int y1, int x2, int y2) {

    }
    for (const auto& p : shape2) {
        s2.insert({p.first + x2, p.second + y2});
    }
    for (const auto& p : s1) {
        if (s2.count(p)) return true;
    }
    return false;
}
