struct Polyomino {
    vector<pair<int, int>> cells;
    int min_x, max_x, min_y, max_y;
    int width, height;
    
    Polyomino(const vector<pair<int, int>>& c) : cells(c) {
        min_x = min_y = INT_MAX;
        max_x = max_y = INT_MIN;
        for (auto& p : cells) {
            min_x = min(min_x, p.first);
            max_x = max(max_x, p.first);
            min_y = min(min_y, p.second);
            max_y = max(max_y, p.second);
        }
        width = max_x - min_x + 1;
        height = max_y - min_y + 1;
    }
};

vector<vector<pair<int, int>>> generate_transforms(const vector<pair<int, int>>& cells) {
    vector<vector<pair<int, int>>> transforms;
    
    // 4 rotations + 4 reflections (8 total)
    for (int r = 0; r < 4; ++r) {
        vector<pair<int, int>> rotated_cells = cells;
        for (int i = 0; i < r; ++i) {
            for (auto& p : rotated_cells) {
                swap(p.first, p.second);
                p.second = -p.second;
            }
        }
        transforms.push_back(rotated_cells);
    }
    
    // Reflection + rotation
    for (int r = 0; r < 4; ++r) {
        vector<pair<int, int>> reflected_cells = cells;
        for (auto& p : reflected_cells) p.first = -p.first;
        for (int i = 0; i < r; ++i) {
            swap(reflected_cells[i].first, reflected_cells[i].second);
            reflected_cells[i].second = -reflected_cells[i].second;
        }
        transforms.push_back(reflected_cells);
    }
    
    return transforms;
}
